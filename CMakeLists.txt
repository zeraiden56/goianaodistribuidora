cmake_minimum_required(VERSION 3.16)
project(DistribuidoraSimulator LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(PkgConfig REQUIRED)
pkg_check_modules(SDL2 REQUIRED IMPORTED_TARGET sdl2)
find_package(OpenGL REQUIRED)
add_library(distribuidora_core src/game.cpp src/player.cpp src/persistence.cpp src/world.cpp src/helper.cpp src/customers.cpp src/computer.cpp)
target_include_directories(distribuidora_core PUBLIC src)
add_executable(distribuidora src/main.cpp src/app.cpp src/render.cpp src/menu.cpp src/audio.cpp src/world_render.cpp src/postprocess.cpp src/interior.cpp src/fixtures.cpp src/customers_render.cpp src/computer_render.cpp src/hud.cpp)
target_link_libraries(distribuidora PRIVATE distribuidora_core PkgConfig::SDL2 OpenGL::GL)
foreach(target distribuidora_core distribuidora)
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
endforeach()
enable_testing()
add_executable(distribuidora_tests tests/tests.cpp)
target_link_libraries(distribuidora_tests PRIVATE distribuidora_core)
target_compile_options(distribuidora_tests PRIVATE -Wall -Wextra -Wpedantic)
add_test(NAME gameplay_and_persistence COMMAND distribuidora_tests)
add_executable(distribuidora_audio_tests tests/audio_tests.cpp src/audio.cpp)
target_include_directories(distribuidora_audio_tests PRIVATE src)
target_link_libraries(distribuidora_audio_tests PRIVATE PkgConfig::SDL2)
target_compile_options(distribuidora_audio_tests PRIVATE -Wall -Wextra -Wpedantic)
add_test(NAME audio COMMAND distribuidora_audio_tests)
set_tests_properties(audio PROPERTIES ENVIRONMENT "SDL_AUDIODRIVER=dummy")

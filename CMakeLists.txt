cmake_minimum_required(VERSION 3.16)
project(DistribuidoraSimulator LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
add_compile_definitions(SDL_MAIN_HANDLED)
if(WIN32)
    # Own main + SDL_MAIN_HANDLED: do not link SDL2main or its entry-point flags.
    find_package(SDL2 CONFIG REQUIRED)
    set(GAME_SDL_TARGET SDL2::SDL2)
    find_package(SDL2_mixer CONFIG REQUIRED)
    set(GAME_MIXER_TARGET SDL2_mixer::SDL2_mixer)
else()
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(SDL2 REQUIRED IMPORTED_TARGET sdl2)
    set(GAME_SDL_TARGET PkgConfig::SDL2)
    pkg_check_modules(SDL2_MIXER REQUIRED IMPORTED_TARGET SDL2_mixer)
    set(GAME_MIXER_TARGET PkgConfig::SDL2_MIXER)
endif()
find_package(PNG REQUIRED)
find_package(OpenGL REQUIRED)
add_library(distribuidora_core src/game.cpp src/supplier.cpp src/player.cpp src/persistence.cpp src/world.cpp src/helper.cpp src/customers.cpp src/computer.cpp)
target_include_directories(distribuidora_core PUBLIC src)
add_executable(distribuidora src/main.cpp src/app.cpp src/render.cpp src/menu.cpp src/audio.cpp src/world_render.cpp src/postprocess.cpp src/interior.cpp src/fixtures.cpp src/customers_render.cpp src/computer_render.cpp src/hud.cpp src/materials.cpp src/traffic_render.cpp src/terminal_render.cpp)
target_link_libraries(distribuidora PRIVATE distribuidora_core ${GAME_SDL_TARGET} ${GAME_MIXER_TARGET} PNG::PNG OpenGL::GL)
target_sources(distribuidora PRIVATE src/startup.cpp src/music.cpp src/music_render.cpp)
target_sources(distribuidora PRIVATE src/controller.cpp src/controller_render.cpp)
foreach(target distribuidora_core distribuidora)
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
endforeach()
enable_testing()
add_executable(distribuidora_tests tests/tests.cpp)
target_link_libraries(distribuidora_tests PRIVATE distribuidora_core)
target_compile_options(distribuidora_tests PRIVATE -Wall -Wextra -Wpedantic)
add_test(NAME gameplay_and_persistence COMMAND distribuidora_tests)
add_executable(distribuidora_audio_tests tests/audio_tests.cpp src/audio.cpp)
target_include_directories(distribuidora_audio_tests PRIVATE src)
target_link_libraries(distribuidora_audio_tests PRIVATE ${GAME_SDL_TARGET} ${GAME_MIXER_TARGET})
target_compile_options(distribuidora_audio_tests PRIVATE -Wall -Wextra -Wpedantic)
add_test(NAME audio COMMAND distribuidora_audio_tests)
set_tests_properties(audio PROPERTIES ENVIRONMENT "SDL_AUDIODRIVER=dummy")
add_executable(distribuidora_music_tests tests/music_tests.cpp src/audio.cpp src/music.cpp)
target_include_directories(distribuidora_music_tests PRIVATE src)
target_link_libraries(distribuidora_music_tests PRIVATE ${GAME_SDL_TARGET} ${GAME_MIXER_TARGET})
target_compile_options(distribuidora_music_tests PRIVATE -Wall -Wextra -Wpedantic)
add_test(NAME music_playlist COMMAND distribuidora_music_tests "${CMAKE_CURRENT_SOURCE_DIR}/src/musicas")
set_tests_properties(music_playlist PROPERTIES ENVIRONMENT "SDL_AUDIODRIVER=dummy")

add_executable(distribuidora_supplier_tests tests/supplier_tests.cpp)
target_link_libraries(distribuidora_supplier_tests PRIVATE distribuidora_core)
target_compile_options(distribuidora_supplier_tests PRIVATE -Wall -Wextra -Wpedantic)
add_test(NAME automatic_supply COMMAND distribuidora_supplier_tests)

add_custom_command(TARGET distribuidora POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_directory
        "${CMAKE_CURRENT_SOURCE_DIR}/src/images" "$<TARGET_FILE_DIR:distribuidora>/images")
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/src/musicas")
    add_custom_command(TARGET distribuidora POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E copy_directory
            "${CMAKE_CURRENT_SOURCE_DIR}/src/musicas" "$<TARGET_FILE_DIR:distribuidora>/musicas")
endif()

add_executable(distribuidora_experience_tests tests/experience_tests.cpp)
target_link_libraries(distribuidora_experience_tests PRIVATE distribuidora_core)
target_compile_options(distribuidora_experience_tests PRIVATE -Wall -Wextra -Wpedantic)
add_test(NAME mixed_orders_and_customer_flow COMMAND distribuidora_experience_tests)
